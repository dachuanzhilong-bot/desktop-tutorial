# PC で流す仕事（bash solver/scripts/pc.sh が上から順に流す）。1 行 = 名前<TAB>コマンド。
# コマンドは solver/ で実行し、$THREADS（スレッド数）・$OUT（出力先、既定 out）・$PRE（プリフロップ解）が使える。
# 結果は $OUT/q/<名前>/ に書くと、小さな JSON が pc-results/ に集められて送られる。済んだ仕事は飛ばす（$OUT/pc-done/<名前>）。
#
# （2026-11-22 から --all-cards の既定は alt-dcfr。済んだ仕事の行には、その時の方式 --update sim-cfr+ を書き足した）
#
# REFACTOR_PLAN 4-3 の材料（1 本 1〜2 分）: As7d2c・49×48 を 30 反復だけ、スレッド数と同時に解くフロップ数を変えて、秒/反復を比べる。
# 49 枚のターンを 24 スレッドで分けると 3 巡目が 1 枚だけになる（17 スレッドでも同じ 3 巡）。2 フロップ × 12 スレッドなら 1 巡あたりの無駄が減る。
4-3-solve-threads-24	./target/release/gen_postflop --all-cards --update sim-cfr+ --iters 30 --pre "$PRE" --out "$OUT/q/4-3-t24" --scenario srp-btn-bb --only-flop 0 --threads 1 --solve-threads 24
4-3-solve-threads-17	./target/release/gen_postflop --all-cards --update sim-cfr+ --iters 30 --pre "$PRE" --out "$OUT/q/4-3-t17" --scenario srp-btn-bb --only-flop 0 --threads 1 --solve-threads 17
4-3-solve-threads-12	./target/release/gen_postflop --all-cards --update sim-cfr+ --iters 30 --pre "$PRE" --out "$OUT/q/4-3-t12" --scenario srp-btn-bb --only-flop 0 --threads 1 --solve-threads 12
4-3-two-flops-12	./target/release/gen_postflop --all-cards --update sim-cfr+ --iters 30 --pre "$PRE" --out "$OUT/q/4-3-two" --scenario srp-btn-bb --flops 2 --threads 2 --solve-threads 12
4-3-general-walk-24	./target/release/gen_postflop --all-cards --update sim-cfr+ --iters 30 --general-walk --pre "$PRE" --out "$OUT/q/4-3-general" --scenario srp-btn-bb --only-flop 0 --threads 1 --solve-threads 24
#
# REFACTOR_PLAN 4-1: As7d2c・srp-btn-bb・全ターン 49 × 全リバー 48 を、交互更新の 2 方式と今の方式で G1 0.3% まで解く。
# .g1.json の looks に、見た反復数と G1 の列が残る（0.5% に届いた所も分かる）。交互更新の 1 反復は 2 走査。
4-1-alt-dcfr	./target/release/gen_postflop --all-cards --g1-target 0.3 --update alt-dcfr --check-every 125 --max-iters 1500 --pre "$PRE" --out "$OUT/q/4-1-alt-dcfr" --scenario srp-btn-bb --only-flop 0 --threads 1 --solve-threads "$THREADS" --checkpoint "$OUT/q-ck/4-1-alt-dcfr" --checkpoint-every 50
4-1-alt-cfr+	./target/release/gen_postflop --all-cards --g1-target 0.3 --update alt-cfr+ --avg-restart 250 --check-every 125 --max-iters 1500 --pre "$PRE" --out "$OUT/q/4-1-alt-cfr+" --scenario srp-btn-bb --only-flop 0 --threads 1 --solve-threads "$THREADS" --checkpoint "$OUT/q-ck/4-1-alt-cfr+" --checkpoint-every 50
4-1-sim-cfr+	./target/release/gen_postflop --all-cards --update sim-cfr+ --g1-target 0.3 --check-every 250 --max-iters 4000 --pre "$PRE" --out "$OUT/q/4-1-sim-cfr+" --scenario srp-btn-bb --only-flop 0 --threads 1 --solve-threads "$THREADS" --checkpoint "$OUT/q-ck/4-1-sim-cfr+" --checkpoint-every 100
#
# REFACTOR_PLAN 3-3-7 の G2'（出荷したデータ。pc-g2.sh、送るのは pc.sh）: 済んだ部分は飛ばすので、途中まで進んでいれば続きから。
3-3-7-g2	NO_PUSH=1 OUT_DIR="$OUT" bash scripts/pc-g2.sh && mkdir -p "$OUT/q/3-3-7-g2" && cp "$OUT/g2.jsonl" "$OUT/q/3-3-7-g2/"
#
# REFACTOR_PLAN 4-1 の G2': DCFR（交互）で 500 反復解いた 3 フロップを、リバーを N = 1,000 で解き直して、配るラインの中で測る。
4-1-g2-alt-dcfr	NO_PUSH=1 OUT_DIR="$OUT/g2-dcfr" SPECS="0 500 - 8 500 - 12 500 -" AVG_RESTART=0 EXTRA_ARGS="--update alt-dcfr" bash scripts/pc-g2.sh && mkdir -p "$OUT/q/4-1-g2-alt-dcfr" && cp "$OUT/g2-dcfr/g2.jsonl" "$OUT/q/4-1-g2-alt-dcfr/"
#
# REFACTOR_PLAN 3-3-7 の G2' の続き（2026-11-22）: 出荷したデータは #0 0.5456%、#8 1.1247%、#12 1.2535% で、2 つが 1.0% を超えた。
# (1) リバーの解き直しを DCFR（交互）にすると、端末の反復数 N = 1,000 のままで下がるか（上流は出荷したデータのまま。1 フロップ 約 40 分）
3-3-7-g2-rdcfr	NO_PUSH=1 OUT_DIR="$OUT" SPECS="8 2500 0.4013 12 3000 0.4777 0 1500 0.4544" EXTRA_ARGS="--resolve-update alt-dcfr" bash scripts/pc-g2.sh && mkdir -p "$OUT/q/3-3-7-g2-rdcfr" && cp "$OUT/g2.jsonl" "$OUT/q/3-3-7-g2-rdcfr/"
# (2) DCFR で解いた上流を DCFR で解き直す（4-1。上流と解き直しの方式をそろえると下がるか。1 フロップ 約 40 分）
4-1-g2-dcfr-rdcfr	for f in 8 12 0; do ./target/release/measure_exploit --resolve-river 1000 --dealt --update alt-dcfr --resolve-update alt-dcfr --pre "$PRE" --scenario srp-btn-bb --only-flop "$f" --iters 500 --solve-threads "$THREADS" --checkpoint "$OUT/g2-dcfr/g2/srp-btn-bb-$f" --checkpoint-every 100 --out "$OUT/g2-dcfr/g2.jsonl" || exit 1; done && mkdir -p "$OUT/q/4-1-g2-dcfr-rdcfr" && cp "$OUT/g2-dcfr/g2.jsonl" "$OUT/q/4-1-g2-dcfr-rdcfr/"
# (3) REFACTOR_PLAN 3-3-4 の決め事「1.0% を超えるフロップがあれば N を上げて測り直す」: 今の解き直し（sim-cfr+）を N = 3,000 で（1 フロップ 約 80 分）
3-3-7-g2-n3000	NO_PUSH=1 OUT_DIR="$OUT" N=3000 SPECS="8 2500 0.4013 12 3000 0.4777" bash scripts/pc-g2.sh && mkdir -p "$OUT/q/3-3-7-g2-n3000" && cp "$OUT/g2.jsonl" "$OUT/q/3-3-7-g2-n3000/"
#
# REFACTOR_PLAN 4-1 の仕上げ（2026-11-22）: 63 フロップを alt-dcfr で作り直す。G1 の目標は 0.3%（0.5% から上げた。alt-dcfr ならほぼ同じ時間）。
# 1 フロップ 5〜10 分、全部で 7〜9 時間の見込み。止まったら同じ 1 行で続きから（済んだフロップは飛ばす）。結果は out/postflop-dcfr。
regen-dcfr	OUT_DIR="$OUT" POSTFLOP_DIR=postflop-dcfr G1_TARGET=0.3 bash scripts/pc-run.sh all; rc=$?; mkdir -p "$OUT/q/regen-dcfr" && (cd "$OUT/postflop-dcfr" && cp --parents */*.g1.json index.json ../q/regen-dcfr/ 2>/dev/null); cp "$OUT/pc-report.txt" "$OUT/q/regen-dcfr/" 2>/dev/null; exit $rc
# 作り直したデータの G2'（srp-btn-bb の 3 フロップ。量産と同じ反復数で作り直し、端末と同じ alt-dcfr・N = 1,000 で解き直す）。約 1 時間
regen-g2	S=""; for f in 0 8 12; do S="$S $f $(sed -n 's/^{"target":[^,]*,"iterations":\([0-9]*\),"g1Pct":\([0-9.]*\),.*/\1 \2/p' "$OUT/postflop-dcfr/srp-btn-bb/$f.g1.json")"; done; echo "SPECS=$S"; NO_PUSH=1 OUT_DIR="$OUT/g2-regen" SPECS="$S" AVG_RESTART=0 EXTRA_ARGS="--update alt-dcfr --resolve-update alt-dcfr" bash scripts/pc-g2.sh && mkdir -p "$OUT/q/regen-g2" && cp "$OUT/g2-regen/g2.jsonl" "$OUT/q/regen-g2/"
# 作り直したデータを GitHub の Release に上げる（gh のログインが要る。pc-publish.sh の先頭）。クラウドの Claude が梱包して差し替える
regen-publish	cd .. && SRC="solver/$OUT/postflop-dcfr" bash solver/scripts/pc-publish.sh
